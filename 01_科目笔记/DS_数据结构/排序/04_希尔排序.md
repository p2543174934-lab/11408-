空间：O(1)
时间：未知（最坏O($n^2$),平均O($nlogn$)）
不稳定
仅适用顺序存储的线性表
```c++
#include <iostream>
#include <vector>
class Solution {
public:
    /**
     * 希尔排序 (Shell Sort)
     * @param nums: 待排序的动态数组
     */
    void shellSort(std::vector<int>& nums) {
        int n = nums.size();
        // 1. 初始增量 gap，逐渐缩小增量直到 gap = 1
        // 这里采用希尔建议的增量序列：n/2, n/4, ..., 1
        for (int gap = n / 2; gap > 0; gap /= 2) {
            // 2. 对每个子序列进行直接插入排序
            // 注意：这里 i 从 gap 开始，实际上是在所有子序列中轮流进行插入
            for (int i = gap; i < n; ++i) {
                int temp = nums[i];
                int j;
                // 3. 经典的插入排序逻辑，但步长从 1 变成了 gap
                // 比较间隔为 gap 的元素
                for (j = i - gap; j >= 0 && nums[j] > temp; j -= gap) {
                    nums[j + gap] = nums[j];
                }
                // 4. 将 temp 放入最终确定的位置
                nums[j + gap] = temp;
            }
        }
    }
};

int main() {
    std::vector<int> data = {8, 9, 1, 7, 2, 3, 5, 4, 6, 0};
    Solution sol;
    sol.shellSort(data);
    std::cout << "希尔排序结果: ";
    for (int x : data) {
        std::cout << x << " ";
    }
    std::cout << std::endl;
    return 0;
}
```
