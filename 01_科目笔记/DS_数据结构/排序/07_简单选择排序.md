时间：恒为O($n^2$)
空间：O(1)
稳定性：不稳定
适用性：顺序存储和链式存储的线性表，尤其适合关键字较少的情况

```c++
#include <iostream>
#include <algorithm> // 包含 std::swap

/**
 * 简单选择排序
 * @param A 待排序数组
 * @param n 数组长度
 */
void SelectSort(int A[], int n) {
    // 一共进行 n-1 趟
    for (int i = 0; i < n - 1; i++) {
        int min = i; // 记录最小元素位置
        
        // 在 A[i...n-1] 中选择最小的元素
        for (int j = i + 1; j < n; j++) {
            if (A[j] < A[min]) {
                min = j; // 更新最小元素位置
            }
        }
        
        // 如果最小元素不在当前位置 i，则进行交换
        if (min != i) {
            std::swap(A[i], A[min]); 
            // 注意：std::swap 内部通常涉及3次移动操作
        }
    }
}

int main() {
    int arr[] = {64, 25, 12, 22, 11};
    int n = sizeof(arr) / sizeof(arr[0]);

    std::cout << "排序前: ";
    for (int i = 0; i < n; i++) std::cout << arr[i] << " ";
    std::cout << std::endl;

    SelectSort(arr, n);

    std::cout << "排序后: ";
    for (int i = 0; i < n; i++) std::cout << arr[i] << " ";
    std::cout << std::endl;

    return 0;
}
```
