- 选一个枢轴，左边元素均小于枢轴，右边均大于等于，此时枢轴元素放在了最终位置，称为一次划分。
- 空间复杂度：最好O($logn$)最坏O(n)平均O(logn)
- 时间复杂度：好O($nlogn$)坏O($n^2$)平均O(nlogn)
	- 提高划分平衡性减少出现最坏情况的发生：三数取中法（取首尾中的中位数），随机选取某个元素
- 稳定性：不稳定
- 适用性：顺序存储的线性表

```c++
#include<iostream>
int Partition(int A[],int low,int high){
	int pivot = A[low];
	while(low < high){
		while(low < high && A[high] >= pivot){
			--high;
		}
		A[low] = A[high];
		while(low < high && A[low] < pivot){
			++low;
		}
		A[high] = A[low];
	}
	A[low] = pivot;
	return low;
}
void QuickSort(int A[],int low,int high){
	if(low < high){
		int pivotpos = Partition(A,low,high);
		QuickSort(A,low,pivotpos - 1);
		QuickSort(A,pivotpos + 1,high);
	}
}
// 测试函数 
int main() { 
	int arr[] = {49, 38, 65, 97, 76, 13, 27, 49}; 
	int n = sizeof(arr) / sizeof(arr[0]); 
	QuickSort(arr, 0, n - 1); 
	std::cout << "快速排序后的数组: "; 
	for (int i = 0; i < n; i++) { 
		std::cout << arr[i] << " "; 
	} 
	return 0; 
	}
```
