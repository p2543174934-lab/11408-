```c++
#include <iostream>
#include <vector>
using namespace std;
/**
 * 计数排序函数
 * @param A 待排序数组（输入）
 * @param B 输出数组（存放排序结果）
 * @param n 待排序元素个数
 * @param k 待排序元素的取值范围 [0, k)
 */
void CountSort(int A[], int B[], int n, int k) {
    // 1. 初始化计数数组 C，并全部清零
    // 使用 vector 或 new 分配，避免栈溢出
    vector<int> C(k, 0);
    // 2. Step1: 统计每个关键字的出现次数
    for (int i = 0; i < n; i++) {
        C[A[i]]++;
    }
    // 3. Step2: 累加计数，确定每个元素在输出数组中的最大位置下标
    // C[i] 保存的是小于或等于 i 的元素个数
    for (int i = 1; i < k; i++) {
        C[i] = C[i] + C[i - 1];
    }
    // 4. Step3: 利用辅助数组 C 实现计数排序（从后往前处理，保证稳定性）
    for (int i = n - 1; i >= 0; i--) {
        // C[A[i]] - 1 就是 A[i] 应该存放的下标位置
        C[A[i]] = C[A[i]] - 1; 
        B[C[A[i]]] = A[i];
    }
}
int main() {
    int A[] = {2, 5, 3, 0, 2, 3, 0, 3}; // 输入数组
    int n = sizeof(A) / sizeof(A[0]);
    int k = 6; // 假设数据范围是 0-5
    
    int B[8]; // 输出数组

    cout << "排序前: ";
    for (int i = 0; i < n; i++) cout << A[i] << " ";

    CountSort(A, B, n, k);

    cout << "\n排序后: ";
    for (int i = 0; i < n; i++) cout << B[i] << " ";
    cout << endl;

    return 0;
}
```
