**堆的应用：堆排序，topk，优先队列**
大顶堆：完全二叉树，根大于等于左右
基于大根堆的堆排序得到的是递增序列
堆排序
空间：O(1)
时间：O(nlogn)
稳定性：不稳定
适用性：顺序存储的线性表

```c++
#include <iostream>
#include <vector>
#include <algorithm> // 包含 std::swap
using namespace std;
/**
 * 将以 k 为根的子树调整为大根堆
 * @param A 数组引用
 * @param k 待调整结点的下标
 * @param len 堆的有效长度
 */
void HeadAdjust(vector<int>& A, int k, int len) {
    A[0] = A[k]; // A[0] 暂存子树的根结点
    for (int i = 2 * k; i <= len; i *= 2) { // 沿 key 较大的子结点向下筛选
        if (i < len && A[i] < A[i + 1]) {
            i++; // 取 key 较大的子结点的下标
        }
        if (A[0] >= A[i]) {
            break; // 筛选结束
        } else {
            A[k] = A[i]; // 将子结点上移
            k = i;       // 继续向下筛选
        }
    }
    A[k] = A[0]; // 被筛选结点放入最终位置
}
/**
 * 建立初始大根堆
 */
void BuildMaxHeap(vector<int>& A, int len) {
    for (int i = len / 2; i > 0; i--) {
        HeadAdjust(A, i, len);
    }
}
/**
 * 堆排序完整逻辑
 */
void HeapSort(vector<int>& A, int len) {
    BuildMaxHeap(A, len); // 初始建堆

    for (int i = len; i > 1; i--) {
        swap(A[i], A[1]);     // 堆顶（最大）与堆底交换
        HeadAdjust(A, 1, i - 1); // 重新调整剩余 i-1 个元素
    }
}
int main() {
    // 下标 0 为占位符，不参与排序，仅作为 HeadAdjust 的暂存单元
    vector<int> data = {0, 53, 17, 78, 9, 45, 65, 87, 32};
    int len = data.size() - 1;

    cout << "排序前: ";
    for (int i = 1; i <= len; i++) cout << data[i] << " ";

    HeapSort(data, len);

    cout << "\n排序后: ";
    for (int i = 1; i <= len; i++) cout << data[i] << " ";
    cout << endl;

    return 0;
}
```
