 ~~~ C++
 //数组方式实现，无哨兵
 #include <iostream>
 #include <vector>
 void InsertSort(int A[],int n){
	 int i,j,temp;
	 for(int i = 1;i < n;i++){
		 if(A[i] < A[i-1]){
			 temp = A[i];
		 }
		 for(j = i - 1;j >= 0 && A[j] > temp;--j){
			 A[j+1] = A[j];
		 }
		 A[j+1] = temp;
	 }
 }
 int main(){
	 int data[] = {45,12,67,33,21,8};
	 int n = sizeof(data) / sizeof(data[0]);
	 InsertSort(data,n);
	 std::cout << "排序后的数组：";
	 for(int i = 0;i < n;i++){
		 std::out << data[i] << " ";
	 }
	 return 0;
 }
 ~~~

``` C++
//链表形式
#include <iostream>
// 定义链表节点结构
struct ListNode {
    int val;
    ListNode *next;
    ListNode(int x) : val(x), next(NULL) {}
};
class Solution {
public:
    ListNode* insertionSortList(ListNode* head) {
        // 如果链表为空或只有一个节点，直接返回
        if (!head || !head->next) return head;
        // 1. 创建一个哑结点，辅助插入操作
        ListNode* dummy = new ListNode(0);
        ListNode* curr = head; // 待排序链表的当前节点
        while (curr) {
            // 记录下一个要处理的节点，防止断链
            ListNode* nextNode = curr->next;
            // 2. 在已排序部分（dummy 后面）寻找插入位置
            // 我们从 dummy 开始往后找，直到找到第一个值大于 curr->val 的节点的前驱
            ListNode* prev = dummy;
            while (prev->next && prev->next->val < curr->val) {
                prev = prev->next;
            }
            // 3. 插入节点：将 curr 插入到 prev 和 prev->next 之间
            curr->next = prev->next;
            prev->next = curr;
            // 4. 继续处理原链表的下一个节点
            curr = nextNode;
        }
        ListNode* sortedHead = dummy->next;
        delete dummy; // 释放哑结点内存
        return sortedHead;
    }
};
// 辅助函数：打印链表
void printList(ListNode* head) {
    while (head) {
        std::cout << head->val << (head->next ? " -> " : "");
        head = head->next;
    }
    std::cout << std::endl;
}
int main() {
    // 构造测试链表: 4 -> 2 -> 1 -> 3
    ListNode* head = new ListNode(4);
    head->next = new ListNode(2);
    head->next->next = new ListNode(1);
    head->next->next->next = new ListNode(3);

    Solution sol;
    std::cout << "排序前: ";
    printList(head);

    head = sol.insertionSortList(head);

    std::cout << "排序后: ";
    printList(head);

    return 0;
}
```
