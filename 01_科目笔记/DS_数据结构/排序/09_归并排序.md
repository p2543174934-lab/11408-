- 核心思想：分治策略，将两个或多个有序序列合并为一个更长的有序序列
- 2路归并：两两归并
- 对于二路归并整个排序过程需要$\lceil log_2n\rceil$
空间复杂度：O(n)
时间复杂度：O(nlogn)
稳定性：稳定
适用性：顺序存储和链式存储的线性表

```c++
#include <iostream>
#include <vector>
using namespace std;
// 辅助数组 B，全局声明或在主函数中分配
int *B; 
/**
 * Merge 函数：将两个有序的部分 [low...mid] 和 [mid+1...high] 归并
 */
void Merge(int A[], int low, int mid, int high) {
    int i, j, k;
    // 1. 将 A 中所有元素复制到辅助数组 B 中
    for (k = low; k <= high; k++) {
        B[k] = A[k];
    }
    // 2. 比较 B 中的两部分，将较小值复制回 A
    for (i = low, j = mid + 1, k = i; i <= mid && j <= high; k++) {
        if (B[i] <= B[j]) {
            A[k] = B[i++]; // 将较小值复制到 A 中
        } else {
            A[k] = B[j++];
        }
    }
    // 3. 处理剩余未复制完的部分
    while (i <= mid)   A[k++] = B[i++];
    while (j <= high)  A[k++] = B[j++];
}
/**
 * MergeSort 函数：递归地对数组进行划分和归并
 */
void MergeSort(int A[], int low, int high) {
    if (low < high) {
        int mid = (low + high) / 2;    // 从中间划分
        MergeSort(A, low, mid);        // 对左半部分归并排序
        MergeSort(A, mid + 1, high);   // 对右半部分归并排序
        Merge(A, low, mid, high);      // 归并
    }
}
int main() {
    int A[] = {49, 38, 65, 97, 76, 13, 27};
    int n = sizeof(A) / sizeof(A[0]);
    
    // 为辅助数组 B 分配空间
    B = new int[n]; 
    
    cout << "排序前: ";
    for (int i = 0; i < n; i++) cout << A[i] << " ";

    // 注意：这里的 low 为 0，high 为 n-1
    MergeSort(A, 0, n - 1);

    cout << "\n排序后: ";
    for (int i = 0; i < n; i++) cout << A[i] << " ";
    cout << endl;

    delete[] B; // 释放内存
    return 0;
}
```
