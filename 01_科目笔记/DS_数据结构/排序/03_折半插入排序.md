空间：O(1)
时间：O($n^2$)
仅适用顺序存储的线性表
稳定
``` c++
#include <iostream>
#include <vector>
class Solution {
public:
    /**
     * 折半插入排序 (Binary Insertion Sort)
     * @param nums: 待排序的动态数组
     */
    void binaryInsertionSort(std::vector<int>& nums) {
        int n = nums.size();
        for (int i = 1; i < n; ++i) {
            int target = nums[i]; // 当前待插入的“摸到的牌”
            int low = 0;
            int high = i - 1;
            // 1. 折半查找：在 nums[0...i-1] 中找到第一个大于 target 的位置
            while (low <= high) {
                int mid = low + (high - low) / 2;
                if (nums[mid] > target) {
                    high = mid - 1; // 目标在左半区
                } else {
                    low = mid + 1;  // 目标在右半区
                }
            }
            // 循环结束时，low 就是 target 应该插入的位置
            // 2. 集中挪位：将 [low, i-1] 区间的元素全部后移一位
            for (int j = i - 1; j >= low; --j) {
                nums[j + 1] = nums[j];
            }
            // 3. 插入元素
            nums[low] = target;
        }
    }
};
int main() {
    std::vector<int> data = {34, 8, 64, 51, 32, 21};
    Solution sol;
    sol.binaryInsertionSort(data);
    std::cout << "折半插入排序结果: ";
    for (int x : data) {
        std::cout << x << " ";
    }
    std::cout << std::endl;
    return 0;
}
```
