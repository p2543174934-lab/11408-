- 写与写，写与读之间存在互斥问题
- 核心思想：设置了一个计数器count用来记录当前正在访问共享文件的进程数，用count的值判断从而做出不同的处理

~~~ java
semaphore rw = 1;//用于实现对共享文件的互斥访问
int count = 0;//记录有几个进程在访问文件
semaphore mutex = 1;//对count的互斥访问
write(){
	while(1){
		P(rw);
		写文件;
		V(rw);
	}
}
reader(){//潜在问题：写进程可能饿死，读进程优先
	while(1){
		P(mutex);
		if(count == 0)
			P(rw);
		count++;
		V(mutex);
		读文件;
		P(mutex);
		count--;
		if(count == 0)
			V(rw);
		V(mutex);
	}
}
~~~

//读写公平法（相对公平的先来先服务）
semaphore rw = 1;//用于实现对共享文件的互斥访问
int count = 0;//记录有几个进程在访问文件
semaphore mutex = 1;//对count的互斥访问
semaphore w = 1;//实现写优先
write(){
	while(1){
		P(w);
		P(rw);
		写文件;
		V(rw);
		V(w);
	}
}
reader(){
	while(1){
		P(w)
		P(mutex);
		if(count == 0)
			P(rw);
		count++;
		V(mutex);
		V(w);
		读文件;
		P(mutex);
		count--;
		if(count == 0)
			V(rw);
		V(mutex);
	}
}
