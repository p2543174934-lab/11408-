- 定义互斥信号量数组chopstick[5]={1，1，1，1，1}.并对哲学家进行0-4编号，哲学家i左边的筷子编号是i，右边筷子编号是(i+1）%5

>[!note]
>如何防止死锁的发生呢？
>- 对哲学家加一下限制条件 -- 最多允许四个哲学家同时进餐，这样保证至少有一个哲学家拿到左右两个筷子
>- 要求奇数号的哲学家先拿左边的筷子，然后再拿右边的筷子，偶数号哲学家刚好相反。保证相邻的两个奇偶号哲学加想吃饭，只有一个可以拿起第一个筷子，避免了占有一个后再等一个
>- 仅当一个哲学家左右两个筷子都可以使用时才允许他抓起筷子



~~~ java
//最多允许四个哲学家同时进餐
semaphore chopstick[5] = {1, 1, 1, 1, 1}; // 5把筷子，初始值为1，表示可用
semaphore count = 4;                      // 增加一个计数信号量，最多允许4个哲学家同时尝试进餐
Pi() { // 第 i 个哲学家 (i = 0, 1, 2, 3, 4)
    while(1) {
        // 1. 申请进餐名额。如果当前已经有4个人在尝试进餐，第5个人会被阻塞在这里
        P(count); 
        // 2. 依次拿起左右两边的筷子
        P(chopstick[i]);         // 拿起左手边的筷子
        P(chopstick[(i+1)%5]);   // 拿起右手边的筷子
        // 3. 进餐
        吃饭;
        // 4. 放下两边的筷子
        V(chopstick[i]);         // 放下左手边的筷子
        V(chopstick[(i+1)%5]);   // 放下右手边的筷子
        // 5. 释放进餐名额，让其他被阻塞的哲学家可以尝试进餐
        V(count); 
        // 6. 思考
        思考;
    }
}
~~~

~~~ java
//奇数号的哲学家先拿左边的筷子，然后再拿右边的筷子，偶数号哲学家刚好相反
semaphore chopstick[5] = {1, 1, 1, 1, 1}; // 5把筷子，初始值为1，表示可用

Pi() { // 第 i 个哲学家 (i = 0, 1, 2, 3, 4)
    while(1) {
        if (i % 2 != 0) { 
            // 奇数号哲学家 (1, 3)：先左后右
            P(chopstick[i]);         // 先申请左手边的筷子
            P(chopstick[(i+1)%5]);   // 再申请右手边的筷子
        } else {
            // 偶数号哲学家 (0, 2, 4)：先右后左
            P(chopstick[(i+1)%5]);   // 先申请右手边的筷子
            P(chopstick[i]);         // 再申请左手边的筷子
        }
        // 进餐
        吃饭;
        // 放下筷子 (放筷子的顺序不会导致死锁，先放哪个都可以)
        V(chopstick[i]);         // 放下左手边的筷子
        V(chopstick[(i+1)%5]);   // 放下右手边的筷子
        // 思考
        思考;
    }
}
~~~

~~~ java
//- 仅当一个哲学家左右两个筷子都可以使用时才允许他抓起筷子
semaphore chopstick[5] = {1, 1, 1, 1, 1}; // 5把筷子
semaphore mutex = 1;                      // 互斥锁，保证拿筷子的过程不可被打断

Pi() { // 第 i 个哲学家 (i = 0, 1, 2, 3, 4)
    while(1) {
        
        // 1. 进入临界区，准备拿筷子
        P(mutex); 
        
        // 2. 依次拿起左右两边的筷子
        P(chopstick[i]);         
        P(chopstick[(i+1)%5]);   
        
        // 3. 筷子都拿到手了，离开临界区，让别人也可以尝试拿筷子
        V(mutex); 

        // 4. 进餐
        吃饭;

        // 5. 放下筷子 (放下筷子不需要加锁)
        V(chopstick[i]);         
        V(chopstick[(i+1)%5]);   

        // 6. 思考
        思考;
    }
}
~~~
